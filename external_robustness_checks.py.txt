# Symbole (positive/reell für physikalische Domäne)
theta, delta, b0, r, sigma, fV = sp.symbols('theta delta b0 r sigma fV', real=True, positive=True)
vpec, H0, rV, Om_m = sp.symbols('vpec H0 rV Om_m', positive=True)

# Falsification-Report Container

def add_to_report(test_name, value_str, threshold_str, status, note=""):

# TEST 1: Faser-Topologie (H^3 Holonomy-Twist → Ob0-Revival?)
# Referenz: App. M.3, M.6, K.3
# =====================================================
try:
    # Triviale S^3: exakte Paritäts-Nullung
    I1_trivial = sp.integrate(sp.cos(theta) * sp.sin(theta)**2, (theta, 0, sp.pi))
    
    # H^3 Proxy: reeller Twist delta ~ rV/rinj (konservativ 0.037)
    I1_hyper = sp.integrate((sp.cos(theta) + delta) * sp.sinh(theta)**2, (theta, 0, sp.pi))
    I1_val = float(sp.re(I1_hyper).subs({delta: 0.037}).evalf())
    
    # Kill-Schwelle: |I1| < 1e-4 (App. I.5.2.1)
    status = "PASS" if abs(I1_val) < 1e-4 else "FAIL"
    add_to_report("1. Faser-Topologie (Parität)", f"{I1_val:.2e}", "< 1.0×10⁻⁴", status, 
                  "Holonomy revival < 10⁻⁵ → Parität stabil")
except Exception as e:
    add_to_report("1. Faser-Topologie", f"SymPy-Error: {str(e)}", "-", "FAIL", "Execution failed")

# =====================================================
# TEST 2: Externe Sensitivität zu ΛCDM-Prior (Ω_m → H0 → θ²)
# Referenz: App. F.1.8, Sec. 2.3 | Planck 2018 σ_Ωm ≈ 0.007
# =====================================================
try:
    # 4-Sigma Variation des Planck-Priors
    sigma_Om = 0.007
    delta_Om_4sigma = 4.0 * sigma_Om  # ≈ 0.028
    
    # QDF Definition (App. G.14)
    theta2 = sp.Rational(2,3) * (vpec / (H0 * rV))**2
    QDF = sp.Rational(2,3) * fV * (1 - fV) * theta2 * (1 + sp.Rational(3,5) * b0**2)
    
    # Sensitivitäts-Proxy: H0 ∝ √Ω_m (flacher Raum) → θ² ∝ 1/Om_m
    QDF_sens = QDF.subs(H0, 67.4 * sp.sqrt(Om_m / 0.3))
    rel_sens = sp.simplify((sp.diff(QDF_sens, Om_m) / QDF_sens) * delta_Om_4sigma)
    rel_sens_val = float(rel_sens.subs({fV:0.68, b0:0.03, vpec:369.82, rV:50, Om_m:0.315}).evalf())
    
    # Kill-Schwelle: |ΔQ/Q| < 15% (robust gegenüber Prior-Drift)
    status = "PASS" if abs(rel_sens_val) < 0.15 else "FAIL"
    add_to_report("2. ΛCDM-Prior Sensitivität (4σ)", f"{rel_sens_val*100:.2f}%", "< ±15%", status, 
                  f"ΔΩ_m = ±{delta_Om_4sigma:.3f} (Planck 4σ) → Robust")
except Exception as e:
    add_to_report("2. ΛCDM-Prior Sensitivität", f"SymPy-Error: {str(e)}", "-", "FAIL", "Execution failed")

# =====================================================
# TEST 3: O(b₀⁴) Closure + Continuum-Limit
# Referenz: App. F.1.2.8, K.4
# =====================================================

    Q2 = sp.Rational(3,5) * b0**2

    ratio = sp.Abs(Q4 / Q2)
    ratio_val = float(ratio.subs({r:50, sigma:35, b0:0.03}).evalf())
    GR_limit = sp.limit(ratio, b0, 0)
    
    # Kill-Schwelle: Ratio < 10⁻³ (App. F.1.2.8.3)
    status = "PASS" if ratio_val < 1e-3 else "FAIL"
    add_to_report("3. O(b₀⁴) Closure & GR-Limit", f"{ratio_val:.2e} (Lim→0: {GR_limit})", "< 1.0×10⁻³", status, 
                  "Truncation sicher; GR exakt wiederherstellbar")
