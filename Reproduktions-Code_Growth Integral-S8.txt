## --- The growth integral and consequences for S8 ---

import numpy as np
from scipy.integrate import quad
from astropy.cosmology import Planck18 as cosmo

# --- Empirische Anker (unverändert) ---
fV = 0.68
v_pec = 369.82  # km/s, Planck 2018 Dipol
H0 = 67.4       # km/s/Mpc, Planck 2018
rV = 50.0       # Mpc, ZOBOV-Median
sigma8_LCDM = 0.811
Omega_m = 0.30

# --- Q_D^F Berechnung ---
theta2 = (2/3) * (v_pec / (H0 * rV))**2
Q0 = (2/3) * fV * (1 - fV) * theta2  # = 6.19e-4

# --- Wachstumsintegral ---
def E(a, Om=Omega_m):
    return np.sqrt(Om / a**3 + (1 - Om))

def integrand(a):
    return 1.0 / (a * E(a)**3)

I_growth, _ = quad(integrand, 1e-3, 1.0, epsabs=1e-12)

# --- S8-Korrektur ---
delta_S8 = -0.5 * (Q0 / (H0/100)**2) * I_growth * sigma8_LCDM  # H0 in 100 km/s/Mpc
S8_FTH = sigma8_LCDM + delta_S8

print(f"I_growth = {I_growth:.6f}")
print(f"Q0/H0^2 = {Q0 / (H0/100)**2:.3e}")
print(f"ΔS₈ = {delta_S8:.3e}")
print(f"S₈^FTH = {S8_FTH:.6f}")
print(f"Rel. Abw. = {100*delta_S8/sigma8_LCDM:.3f}%")
