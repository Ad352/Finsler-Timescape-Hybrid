class K7_ExplicitCurvatureValidator:
    def __init__(self, tol=1e-6, max_iter=20, damping=0.7):
        # Empirical Anchors (FTH v2.6.1, Appendix F & G)
        self.f_V     = 0.68            # SDSS-ZOBOV DR12
        self.v_pec   = 369.82          # km/s (Planck 2018 dipole)
        self.r_V     = 50.0            # Mpc (ZOBOV median)
        self.Omega_b_h2 = 0.0224       # BBN anchor (Cooke+2018)
        self.b0_0    = 1.232e-3        # b_CMB = v_pec/c (IR fixed point)
        self.alpha2  = 3/5             # S3 angular averaging constant
        
        # Geometric closure coefficient (derived from void-wall junction topology)
        # C_topo ~ 22.0 emerges from hyperbolic void interior scaling at f_V=0.68
        self.C_topo  = 22.0            
        
        # Pre-compute J.11-decoupled curvature scale (km/s/Mpc)^2
        self.K_void  = self.C_topo * (1 - self.f_V) / self.f_V * (self.v_pec / self.r_V)**2
        
        # Numerical settings
        self.tol      = tol
        self.max_iter = max_iter
        self.damping  = damping

    def _qdf(self, H):
        """Nonlinear backreaction Q_D,F(H) with Riccati-evolved b0(z=0)"""
        sigma2 = (self.v_pec / (H * self.r_V))**2
        Q0 = (2/3) * self.f_V * (1 - self.f_V) * sigma2
        return Q0 * (1 + self.alpha2 * self.b0_0**2)

    def iteration_map(self, H_old):
        """F.1.9.4 + Variant 1: H_new = sqrt(R_BBN + K_void + Q_DF(H_old)/3)"""
        R_BBN = 1e4 * self.Omega_b_h2  # Constant baryon term (km/s/Mpc)^2
        Q_DF  = self._qdf(H_old)
        return np.sqrt(R_BBN + self.K_void + Q_DF / 3.0)

    def run_k7_iteration(self, H_seed=68.0):
        """Fixed-point iteration with under-relaxation for stability"""
        H_current = H_seed
        history = []
        print("🔄 Starting K-7 iteration with explicit void curvature (J.11-decoupled)...\n")
        
        for i in range(1, self.max_iter + 1):
            H_new = self.iteration_map(H_current)
            rel_err = abs(H_new - H_current) / H_current
            history.append({"iter": i, "H_old": H_current, "H_new": H_new, "rel_err": rel_err})
            print(f"Iter {i:2d} | H0 = {H_current:6.4f} -> {H_new:6.4f} km/s/Mpc | Δ = {rel_err:.2e}")
            
            if rel_err < self.tol:
                self.H_FTH = H_new
                print("\n✅ CONVERGENCE ACHIEVED.")
                return history
            
            H_current = (1 - self.damping) * H_current + self.damping * H_new
            
        raise RuntimeError(f"❌ Failed to converge in {self.max_iter} steps. Last Δ = {rel_err:.2e}")

    def evaluate_k7(self):
        """K-7 Kill-Switch evaluation against Planck & SH0ES"""
        sigma_Planck = abs(self.H_FTH - 67.4) / 0.5
        sigma_SH0ES  = abs(self.H_FTH - 73.0) / 1.0
        # K-7 triggers if OUTSIDE BOTH 3σ bands
        kill = (sigma_Planck > 3) and (sigma_SH0ES > 3)
        return {"H_FTH": self.H_FTH, "σ_Planck": sigma_Planck, 
                "σ_SH0ES": sigma_SH0ES, "KILL": kill, "K_void": self.K_void}
